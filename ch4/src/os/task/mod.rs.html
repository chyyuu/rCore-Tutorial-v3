27/// The task manager, where all the tasks are managed.
28///
29/// Functions implemented on `TaskManager` deals with all task state transitions
30/// and task context switching. For convenience, you can find wrappers around it
31/// in the module level.
32///
33/// Most of `TaskManager` are hidden behind the field `inner`, to defer
34/// borrowing checks to runtime. You can see examples on how to use `inner` in
35/// existing functions on `TaskManager`.
36pub struct TaskManager {
37    /// total number of tasks
38    num_app: usize,
39    /// use inner value to get mutable access
40    inner: UPSafeCell<TaskManagerInner>,
41}
42
43/// The task manager inner in 'UPSafeCell'
44struct TaskManagerInner {
45    /// task list
46    tasks: Vec<TaskControlBlock>,
47    /// id of current `Running` task
48    current_task: usize,
49}
50
51lazy_static! {
52    /// a `TaskManager` global instance through lazy_static!
53    pub static ref TASK_MANAGER: TaskManager = {
54        println!("init TASK_MANAGER");
55        let num_app = get_num_app();
56        println!("num_app = {}", num_app);
57        let mut tasks: Vec<TaskControlBlock> = Vec::new();
58        for i in 0..num_app {
59            tasks.push(TaskControlBlock::new(get_app_data(i), i));
60        }
61        TaskManager {
62            num_app,
63            inner: unsafe {
64                UPSafeCell::new(TaskManagerInner {
65                    tasks,
66                    current_task: 0,
67                })
68            },
69        }
70    };
71}
72
73impl TaskManager {
74    /// Run the first task in task list.
75    ///
76    /// Generally, the first task in task list is an idle task (we call it zero process later).
77    /// But in ch4, we load apps statically, so the first task is a real app.
78    fn run_first_task(&self) -> ! {
79        let mut inner = self.inner.exclusive_access();
80        let next_task = &mut inner.tasks[0];
81        next_task.task_status = TaskStatus::Running;
82        let next_task_cx_ptr = &next_task.task_cx as *const TaskContext;
83        drop(inner);
84        let mut _unused = TaskContext::zero_init();
85        // before this, we should drop local variables that must be dropped manually
86        __switch(&mut _unused as *mut _, next_task_cx_ptr);
87        panic!("unreachable in run_first_task!");
88    }
89
90    /// Change the status of current `Running` task into `Ready`.
91    fn mark_current_suspended(&self) {
92        let mut inner = self.inner.exclusive_access();
93        let cur = inner.current_task;
94        inner.tasks[cur].task_status = TaskStatus::Ready;
95    }
96
97    /// Change the status of current `Running` task into `Exited`.
98    fn mark_current_exited(&self) {
99        let mut inner = self.inner.exclusive_access();
100        let cur = inner.current_task;
101        inner.tasks[cur].task_status = TaskStatus::Exited;
102    }
103
104    /// Find next task to run and return task id.
105    ///
106    /// In this case, we only return the first `Ready` task in task list.
107    fn find_next_task(&self) -> Option<usize> {
108        let inner = self.inner.exclusive_access();
109        let current = inner.current_task;
110        (current + 1..current + self.num_app + 1)
111            .map(|id| id % self.num_app)
112            .find(|id| inner.tasks[*id].task_status == TaskStatus::Ready)
113    }
114
115    /// Get the current 'Running' task's token.
116    fn get_current_token(&self) -> usize {
117        let inner = self.inner.exclusive_access();
118        inner.tasks[inner.current_task].get_user_token()
119    }
120
121    /// Get the current 'Running' task's trap contexts.
122    fn get_current_trap_cx(&self) -> &'static mut TrapContext {
123        let inner = self.inner.exclusive_access();
124        inner.tasks[inner.current_task].get_trap_cx()
125    }
126
127    /// Switch current `Running` task to the task we have found,
128    /// or there is no `Ready` task and we can exit with all applications completed
129    fn run_next_task(&self) {
130        if let Some(next) = self.find_next_task() {
131            let mut inner = self.inner.exclusive_access();
132            let current = inner.current_task;
133            inner.tasks[next].task_status = TaskStatus::Running;
134            inner.current_task = next;
135            let current_task_cx_ptr = &mut inner.tasks[current].task_cx as *mut TaskContext;
136            let next_task_cx_ptr = &inner.tasks[next].task_cx as *const TaskContext;
137            drop(inner);
138            // before this, we should drop local variables that must be dropped manually
139            __switch(current_task_cx_ptr, next_task_cx_ptr);
140            // go back to user mode
141        } else {
142            println!("All applications completed!");
143            use crate::board::QEMUExit;
144            crate::board::QEMU_EXIT_HANDLE.exit_success();
145        }
146    }
147}

149/// Run the first task in task list.
150pub fn run_first_task() {
151    TASK_MANAGER.run_first_task();
152}
153
154/// Switch current `Running` task to the task we have found,
155/// or there is no `Ready` task and we can exit with all applications completed
156fn run_next_task() {
157    TASK_MANAGER.run_next_task();
158}
159
160/// Change the status of current `Running` task into `Ready`.
161fn mark_current_suspended() {
162    TASK_MANAGER.mark_current_suspended();
163}
164
165/// Change the status of current `Running` task into `Exited`.
166fn mark_current_exited() {
167    TASK_MANAGER.mark_current_exited();
168}
169
170/// Suspend the current 'Running' task and run the next task in task list.
171pub fn suspend_current_and_run_next() {
172    mark_current_suspended();
173    run_next_task();
174}
175
176/// Exit the current 'Running' task and run the next task in task list.
177pub fn exit_current_and_run_next() {
178    mark_current_exited();
179    run_next_task();
180}
181
182/// Get the current 'Running' task's token.
183pub fn current_user_token() -> usize {
184    TASK_MANAGER.get_current_token()
185}

187/// Get the current 'Running' task's trap contexts.
