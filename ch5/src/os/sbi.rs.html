1//! SBI call wrappers
2#![allow(unused)]
3
4use core::arch::asm;
5
6// Legacy SBI call numbers (for compatibility)
7const SBI_CONSOLE_PUTCHAR: usize = 1;
8const SBI_CONSOLE_GETCHAR: usize = 2;
9
10// SBI Extension IDs
11const SBI_EXT_TIMER: usize = 0x54494D45;
12const SBI_EXT_SRST: usize = 0x53525354;
13
14/// General SBI call with extension ID and function ID
15#[inline(always)]
16fn sbi_call(eid: usize, fid: usize, arg0: usize, arg1: usize, arg2: usize) -> usize {
17    let mut ret;
18    unsafe {
19        asm!(
20            "ecall",
21            inlateout("x10") arg0 => ret,
22            in("x11") arg1,
23            in("x12") arg2,
24            in("x16") fid,
25            in("x17") eid,
26        );
27    }
28    ret
29}
30
31/// Set timer using the SBI Timer Extension
32pub fn set_timer(timer: u64) {
33    #[cfg(target_pointer_width = "64")]
34    sbi_call(SBI_EXT_TIMER, 0, timer as usize, 0, 0);
35    #[cfg(target_pointer_width = "32")]
36    sbi_call(SBI_EXT_TIMER, 0, timer as usize, (timer >> 32) as usize, 0);
37}
38
39/// Use SBI call to put a character to console
40pub fn console_putchar(c: usize) {
41    sbi_call(SBI_CONSOLE_PUTCHAR, 0, c, 0, 0);
42}
43
44/// Use SBI call to get a character from console
45pub fn console_getchar() -> usize {
46    sbi_call(SBI_CONSOLE_GETCHAR, 0, 0, 0, 0)
47}
48
49/// Use SBI call to shutdown the system
50pub fn shutdown() -> ! {
51    sbi_call(SBI_EXT_SRST, 0, 0, 0, 0);
52    panic!("It should shutdown!");
53}
